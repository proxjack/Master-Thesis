\section{Motivation and Objectives}

% ---------------------------------------------------------------- 2
\begin{frame}{Why energy harvesting?}
    \begin{itemize}
        \item WSN and IoT nodes are everywhere, \alert{collecting data} for many
              different purposes.
        \item Batteries are costly to maintain, have a finite life and are often
              \alert{impossible to replace}, and they are bad for the environment.
        \item With \alert{energy harvesting} the node powers itself from the
              energy already available around it.
    \end{itemize}
    \vfill
    \centering
    \includegraphics[width=0.74\textwidth]{VREH_basic.png}
    \vfill
\end{frame}

% ---------------------------------------------------------------- 3
\begin{frame}{The variable reluctance principle}
    \begin{columns}[T]
        \column{0.44\textwidth}
            \small
            \begin{itemize}
                \item Magnet and coil \alert{stand still}; only the toothed wheel
                      rotates.
                \item The passing teeth modulate the flux and induce an EMF.
            \end{itemize}
            \vspace{1.5ex}
            \textbf{Advantages}
            \begin{itemize}
                \item No moving electrical parts
                \item Scalable to any shaft size
                \item Very small counter torque
            \end{itemize}
        \column{0.54\textwidth}
            \centering
            \includegraphics[width=\textwidth]{VR_theory.pdf}\\[1ex]
            {\scriptsize Pickup unit facing the rotating wheel.}
    \end{columns}
\end{frame}

\section{Modelling and Proposed Topologies}

% ---------------------------------------------------------------- 7
\begin{frame}{Coil model and matching}
    \begin{columns}[T]
        \column{0.50\textwidth}
            \small
            \begin{itemize}
                \item Each coil is a source in series with $R_\text{Coil}$ and
                      $L_\text{Coil}$.
                \item At low speed a \alert{resistive} load is enough.
                \item At higher speed $\omega_e L_\text{Coil}$ breaks the matching.
                \item A series capacitor $C_m$ cancels it
            \end{itemize}
            \begin{equation*}
                C_m = \frac{1}{\omega_e^{2}\,L_\text{Coil}}
            \end{equation*}
            \centering
            {\small valid at one frequency at a time.}
        \column{0.48\textwidth}
            \centering
            \includegraphics[width=0.95\textwidth]{Coil_model.pdf}\\[2ex]
            \includegraphics[width=0.85\textwidth]{Matching_condition.pdf}
    \end{columns}
\end{frame}

% ---------------------------------------------------------------- 8
\begin{frame}{The six real phase voltages}
    \begin{columns}[T]
        \column{0.60\textwidth}
            \centering
            \includegraphics[width=\textwidth]{OCV_100_focus.pdf}
        \column{0.38\textwidth}
            \begin{itemize}
                \item Measured at $100\ \text{rpm}$.
                \item The $60^\circ$ shift is clearly visible.
                \item \alert{Not} clean sine waves, and the six units are not
                      identical.
            \end{itemize}
    \end{columns}
\end{frame}

% ---------------------------------------------------------------- 9
\begin{frame}{From waveform to simulation}
    \begin{columns}[T]
        \column{0.42\textwidth}
            \small
            \begin{itemize}
                \item RMS voltage computed against speed.
                \item Each source replaced by an \alert{equivalent sinusoid},
                      scaled by the crest factor.
                \item Same power as the real waveform.
            \end{itemize}
            \vspace{0.5ex}
            Every phase is then one source, delayed by $60^\circ$ from the previous
            one,
            \begin{equation*}
                e_k(t) = E_{pk}\sin\!\left(\omega_e t - k\,\frac{\pi}{3}\right)
            \end{equation*}
            \centering
            with $k = 0, 1, \dots, 5$.
        \column{0.56\textwidth}
            \centering
            \includegraphics[width=\textwidth]{Voltage_comp.pdf}\\[0.5ex]
            {\scriptsize FEM, raw measured peak, crest-factor corrected.}
    \end{columns}
\end{frame}

% ---------------------------------------------------------------- 10
\begin{frame}{Strategy 1 - FWR bridges}
    \begin{columns}[T]
        \column{0.50\textwidth}
            \small
            \alert{Two three-phase groups} ($k=0,2,4$ and $k=1,3,5$), each on a
            standard bridge.
            \vspace{1ex}
            \begin{itemize}
                \item The \textbf{Star (Y)} gives a line voltage $\sqrt{3}$ times
                      the phase EMF, so the two-diode drop costs less.
                \item The \textbf{Delta ($\Delta$)} gives the coil EMF itself, with
                      no gain but no floating node.
            \end{itemize}
            \vspace{1ex}
            The Star should give the \alert{higher open-circuit voltage}.
        \column{0.48\textwidth}
            \centering
            \includegraphics[width=0.92\textwidth]{FWR.pdf}\\[1ex]
            {\scriptsize Two diodes always in the current path.}
    \end{columns}
\end{frame}

% ---------------------------------------------------------------- 11
\begin{frame}{Strategy 2 - NVC bridges}
    \begin{columns}[T]
        \column{0.50\textwidth}
            \small
            \alert{Three two-phase groups}, pairing phases $180^\circ$ apart in
            \textbf{anti-series}
            \begin{equation*}
                v_A - v_B = e_k - e_{k+3} = 2\,e_k
            \end{equation*}
            ideally double the voltage.
            \vspace{1ex}
            \begin{itemize}
                \item Four MOSFETs with \alert{cross-coupled gates}, so no gate
                      driver is needed.
                \item They bypass the diodes, trading the fixed drop for a small
                      $R_\text{on}$.
                \item Below $V_\text{off}\approx1.2\ \text{V}$ it falls back to the
                      diodes.
            \end{itemize}
        \column{0.48\textwidth}
            \centering
            \includegraphics[width=0.95\textwidth]{NVC.pdf}\\[1ex]
            {\scriptsize NVC for one anti-series pair.}
    \end{columns}
\end{frame}

% ---------------------------------------------------------------- 18
\begin{frame}{Simulation vs measurement}
    Measured power differs from simulation by \alert{8\% for the Star, up to almost
    30\% for the NVC}.
    \vfill
    \begin{table}
        \centering
        \small
        \begin{tabular}{lccc}
            \toprule
            & Delta & Star & NVC\\
            \midrule
            Mean deviation of $P_O$ from simulation & 16\% & 8\% & \alert{30\%}\\
            \bottomrule
        \end{tabular}
    \end{table}
    \vfill
    Why?
    \begin{itemize}
        \item The ceramic $C_m$ \alert{lose capacitance} under the real voltage,
              which grows with speed.
        \item Component tolerances and the parasitic resistance of the bridges.
        \item Sources that are not perfectly sinusoidal.
    \end{itemize}
\end{frame}

\section{Conclusions}

% ---------------------------------------------------------------- 20
\begin{frame}{Conclusions and future work}
    \begin{block}{Conclusions}
        \begin{itemize}
            \item \alert{NVC and Star} are the two strongest candidates, one per
                  end of the speed range; Delta is always the weakest.
            \item NVC and Delta fit any PMIC tested; the Star needs protection at
                  high speed, its open-circuit voltage can damage the input.
        \end{itemize}
    \end{block}
    \begin{alertblock}{Limitation}
        The harvester prototype is 3D-printed and sensitive to vibration at high
        speed, which makes the measurements harder to repeat.
    \end{alertblock}
    \begin{block}{Future work}
        Adaptive compensation tracking $C_m$ with speed, and a full hardware test
        up to a real PMIC, to measure the efficiency from harvester to load.
    \end{block}
\end{frame}
